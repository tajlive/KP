FIX:
  Kjør dette i Supabase SQL Editor (dashboard.supabase.com → SQL Editor):

  -- Oppdater calculate_daily_loss til å bruke riktig tabell
  CREATE OR REPLACE FUNCTION calculate_daily_loss(
    p_user_id UUID,
    p_session_id UUID DEFAULT NULL
  )
  RETURNS NUMERIC
  LANGUAGE SQL
  STABLE
  SECURITY DEFINER
  AS $$
    SELECT COALESCE(
      SUM(ABS(COALESCE(pnl, 0))),
      0
    )
    FROM auto_trading_trades
    WHERE user_id = p_user_id
      AND (p_session_id IS NULL OR session_id = p_session_id)
      AND status = 'closed'
      AND executed_at >= NOW() - INTERVAL '24 hours'
      AND pnl < 0;
  $$;

  -- Gi execute-rettigheter til authenticated brukere
  GRANT EXECUTE ON FUNCTION calculate_daily_loss(UUID, UUID)
    TO authenticated;
