# Systematisk i18n — Slutt på MISSING_MESSAGE for alltid

> **For Hermes:** Følg planen punkt for punkt. Ingen implementering før planen er godkjent.

**Mål:** Null `MISSING_MESSAGE`-feil i produksjon. Automatisk validering ved commit som fanger alle manglende i18n-nøkler før deploy.

**Arkitektur:** Nytt node-script `scripts/validate-i18n-usage.js` som scanner ALLE `.tsx`/`.ts`-filer for `t("key")`-kall, resolver namespace fra `useTranslations("ns")`, og sjekker at hver nøkkel finnes i `messages/no.json`. Scriptet legges til pre-commit gate 1.

**Tech Stack:** Node.js, regex (ingen eksterne avhengigheter), JSON, TypeScript

**Rotårsak:** `validate-i18n.js` sjekker kun at `no.json` ↔ `en.json` har samme struktur. Den sjekker **aldri** om komponentenes `t("key")`-kall har tilsvarende nøkkel i messages. Resultat: nye `t("key")` legges til uten messages-nøkkel → produksjonsfeil.

### Task 2: Opprett `scripts/validate-i18n-usage.js`

**Objektiv:** Lag et node-script som scanner koden for `t("key")`-kall og validerer mot `no.json`. Skal kunne kjøres som `npm run i18n:usage`.

**Filer:**
- Create: `scripts/validate-i18n-usage.js`

**Steg 1: Skriv scriptet**

```javascript
#!/usr/bin/env node
/**
 * i18n Usage Validation
 * Sjekker at alle t("key")-kall i kildekoden har tilsvarende nøkkel i messages/no.json
 */
import fs from 'fs';
import path from 'path';
import { fileURLToPath } from 'url';

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);
const SRC_DIR = path.join(__dirname, '..', 'src');
const MSG_FILE = path.join(__dirname, '..', 'messages', 'no.json');

function loadJSON(fp) {
  return JSON.parse(fs.readFileSync(fp, 'utf-8'));
}

function buildKeyMap(obj, prefix = '') {
  const keys = new Set();
  for (const [k, v] of Object.entries(obj)) {
    const full = prefix ? `${prefix}.${k}` : k;
    keys.add(full);
    if (typeof v === 'object' && v !== null && !Array.isArray(v)) {
      for (const sk of buildKeyMap(v, full)) keys.add(sk);
    }
  }
  return keys;
}

function keyExists(messages, fullKey) {
  const parts = fullKey.split('.');
  let cur = messages;
  for (const p of parts) {
    if (typeof cur !== 'object' || cur === null || !(p in cur)) return false;
    cur = cur[p];
  }
  return true;
}

function walkDir(dir) {
  const files = [];
  for (const entry of fs.readdirSync(dir, { withFileTypes: true })) {
    const fp = path.join(dir, entry.name);
    if (entry.isDirectory()) {
      if (!['node_modules', '.next', '.git'].includes(entry.name)) {
        files.push(...walkDir(fp));
      }
    } else if (entry.isFile() && /\.(tsx|ts)$/.test(entry.name)) {
      files.push(fp);
    }
  }
  return files;
}

function findKeyCalls(content) {
  // Regex that captures t("key") and t(`key`) patterns
  const regex = /t\([`'"]([^`'"]+)[`'"]\)/g;
  const keys = [];
  let match;
  while ((match = regex.exec(content)) !== null) {
    keys.push(match[1]);
  }
  return keys;
}

function main() {
  const messages = loadJSON(MSG_FILE);
  const sourceFiles = walkDir(SRC_DIR);
  
  let totalMissing = 0;
  const missingByNs = {};
  let checkedFiles = 0;
  
  for (const fp of sourceFiles) {
    const content = fs.readFileSync(fp, 'utf-8');
    const lines = content.split('\n');
    
    // Find namespace
    let ns = null;
    for (const line of lines) {
      const m = line.match(/useTranslations\("([^"]+)"\)/);
      if (m) { ns = m.group(1); break; }
    }
    if (!ns) continue; // Skip files without useTranslations
    
    checkedFiles++;
    const keys = findKeyCalls(content);
    
    for (const key of keys) {
      const fullKey = `${ns}.${key}`;
      if (!keyExists(messages, fullKey)) {
        totalMissing++;
        if (!missingByNs[ns]) missingByNs[ns] = [];
        missingByNs[ns].push({ file: path.relative(SRC_DIR, fp), key, fullKey });
      }
    }
  }
  
  console.log(`\n📊  Sjekket ${checkedFiles} filer med useTranslations\n`);
  
  if (totalMissing === 0) {
    console.log('✅  Alle t("key")-kall har tilsvarende nøkkel i messages/no.json');
    process.exit(0);
  }
  
  console.log(`❌  Fant ${totalMissing} manglende nøkler:\n`);
  
  for (const [ns, items] of Object.entries(missingByNs).sort()) {
    console.log(`  [${ns}]`);
    for (const item of items) {
      console.log(`    - ${item.fullKey}`);
      console.log(`      ${item.file}`);
    }
    console.log('');
  }
  
  process.exit(1);
}

main();
```

## Risiko og avveininger

| Risiko | Sannsynlighet | Tiltak |
|--------|--------------|--------|
| Regex fanger falske positive | Medium | `t()`-regex er spesifikk nok; test mot kjente edge-cases |
| Scriptet tar >5s på stor kodebase | Lav | `walkDir` er synkron og rask (~1s for 1500 filer) |
| Pre-commit blir tregere | Lav | Scriptet kjører parallelt med andre checker i gate 1 |
| Noen `t()`-kall bruker dynamiske nøkler | Medium | Håndteres av regex — dynamiske nøkler (`t(\`key.\${var}\`)`) er sjeldne |
