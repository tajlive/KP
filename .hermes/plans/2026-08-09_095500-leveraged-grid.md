// leveraged-grid.ts
export class LeveragedGridStrategy extends GridBotStrategy {
  private leverage: number;
  private liquidationPrice: number;
  private marginRatio: number;
  
  constructor(config: LeveragedGridConfig) {
    super(config);
    this.leverage = config.leverage || 2;
    this.marginRatio = 1 / this.leverage;
  }
  
  evaluate(input: StrategyInput): StrategySignal {
    const baseSignal = super.evaluate(input);
    
    // Sjekk liquidation-avstand
    if (this.isNearLiquidation(input.currentPrice)) {
      return { action: 'hold', reason: 'near_liquidation' };
    }
    
    // Juster posisjonsstørrelse for leverage
    return {
      ...baseSignal,
      quantity: baseSignal.quantity * this.leverage,
      leverage: this.leverage,
    };
  }
  
  private isNearLiquidation(currentPrice: number): boolean {
    const distance = Math.abs(currentPrice - this.liquidationPrice) / currentPrice;
    return distance < 0.05; // 5% buffer
  }
}
```

// liquidation-guard.ts
export class LiquidationGuard {
  private warningThreshold = 0.10; // 10% fra liquidation
  private criticalThreshold = 0.05; // 5% fra liquidation
  
  check(position: Position, currentPrice: number): LiquidationStatus {
    const distance = this.calculateDistance(position, currentPrice);
    
    if (distance < this.criticalThreshold) {
      return { status: 'critical', action: 'close_position' };
    }
    if (distance < this.warningThreshold) {
      return { status: 'warning', action: 'reduce_position' };
    }
    return { status: 'safe', action: 'none' };
  }
}
```
